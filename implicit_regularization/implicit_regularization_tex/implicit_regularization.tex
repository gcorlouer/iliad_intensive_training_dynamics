\begin{frame}{Definition}
  $\Omega$: set of parameters; $L(\theta)$: population loss of a neural network $f(\theta,x)$ with input data $x$.

  \vspace{0.4cm}

  Gradient descent:
  \[
    \Delta \theta_k = -\eta\nabla L(\theta_k)
  \]

  Convergence to $\theta^{\star}\in \Omega^{\star}= \operatorname{argmin}_{\theta} L(\theta)$

  \vspace{0.4cm}

  Implicit regularization: there exists a complexity function $R[\theta]$:
  \[
    \theta^{\star} = \min_{\theta\in \Omega^{\star}} R[\theta]
  \]

  GD prefers solutions within certain complexity classes
\end{frame}

\begin{frame}{Explicit regularization}
  Define regularized loss and optimize it:
  \[
    \mathcal{L}(\theta, x) = L(\theta,x) + \lambda R(\theta)
  \]

  Example weight decay:
  \[
    R = ||\theta||_2^2
  \]
\end{frame}

\begin{frame}{Why do we care?}
  \begin{itemize}
    \item Contrary to naive statistics, deep neural networks (DNNs) generalize surprisingly well
    \item Implicit regularization must occur
    \item SLT tells us a story for Bayesian learning
    \item We want a theory of implicit regularization for SGD
    \item Implicit regularization from scale (depth), loss, optimizer, initialization,...
  \end{itemize}
\end{frame}

\begin{frame}{Implications for alignment}
  \begin{itemize}
    \item Alignment is also a question about implicit regularization
    \item Observe policy $\pi(\theta,x)$: the policy does well on the reward and looks aligned in behavioural tests
    \item But perhaps $\pi(\theta,x)$ is scheming (misaligned) because it is favoured by training
    \item If there is no strong regularization for honest solutions, then schemers are more likely
    \item Hope 1: Understanding implicit bias helps us detect scheming
    \item Hope 2: Design a training process that selects against schemers
  \end{itemize}
\end{frame}

\begin{frame}{Deep Linear Networks}
  \small
  Function (multilinear in parameters):
  \[
    \begin{aligned}
      f: \ & \mathbb{R}^{d_0} \times \prod_{l=0}^{L-1}\mathbb{R}^{d_l \times d_{l+1}}  \to \mathbb{R}^{d_L} \\
         & (x,W_1,...,W_L) \mapsto W_L...W_1 x = Wx
    \end{aligned}
  \]

  Function $W$ (student) learns $M$ (teacher)

  We train on data generated by $y=Mx$ with quadratic loss and input covariance matrix $\Sigma_X = I$:
  \[
    \mathcal{L}_N(\theta; D_N) =
    \frac{1}{2N} \sum_{i=1}^{N} ||y_i - W_L...W_1x_i||_2^2 =
    \frac{1}{2N} \sum_{i=1}^{N} ||Mx_i - Wx_i||_2^2
    \to_N ||(M - W)||^2_F
  \]

  Non-convex, high-dimensional loss landscape
\end{frame}

\begin{frame}{Symmetries}
  \small
  Let $GL_{\textbf{d}}:=\Pi_{l=L}^0 GL_{d_l} = GL_{d_L}\times GL_{h} \times GL_{d_0}$ act on $\Omega$ via:
  \[
    \theta=(W_1,\dots,W_L)\mapsto
    (g_1W_1g_0^{-1},\; g_2W_2g_1^{-1},\;\dots,\; g_L W_L g_{L-1}^{-1})
  \]

  Define $\mu(\theta) = W$
  \[
    \mu(g\cdot \theta)=g_LWg_0^{-1};\quad \mu(g_h\cdot \theta) = \mu(\theta)
  \]

  Let $\theta\in \pi^{-1}(W)$, the orbit $\mathcal{O}_{\theta}:=G_{\mathrm{h}}\cdot \theta$ is a subspace of the fiber $\mu^{-1}(W)$
\end{frame}

\begin{frame}{Global minima}
  \small
  \begin{itemize}
    \item If $d_l \geq d_L$, there is a unique solution in function space for $W^{\star} = \Sigma_{YX}=M$ (OLS), with $\Sigma_{YX}$ the input-output covariance
    \item A whole space of parameters maps to the solution (degenerate)
    \item Key geometric invariants under symmetries: rank patterns $r_{ij}=\operatorname{rank}(W_i...W_j)$
    \item Can stratify the degenerate parameters mapping to global minima into $O_{r_{ij}}$
    \item Gives a combinatorial description of global minima
    \item Can be used to compute the dimension of global minima (link with RLCT)
    \item See more at this \href{https://arxiv.org/abs/2411.19920}{paper from Simon Pepin Lehalleur and Rich{\'a}rd Rim{\'a}nyi}
    \item Informally: the rank patterns quantify simplicity among degenerate solutions
  \end{itemize}
\end{frame}

\begin{frame}{Saddle points}
  In DLNs, if $d_l \geq d_L$, with full-rank and non-degenerate data:
  \begin{itemize}
    \item \textbf{All critical points are either saddles or global minima}
    \item Saddles can be strict or non-strict
    \item They are also often highly degenerate (many flat directions)
    \item This is generic in DNNs
  \end{itemize}

  Work is in progress to get the combinatorics of saddles with quiver representation theory
\end{frame}

\begin{frame}{First order critical points}
  \small
  Assume $d_l\geq d_L$ and well-behaved data. Let $\theta$ be a critical point of the population loss $\mathcal{L}$.

  From \href{https://arxiv.org/abs/2107.13289}{Achour et al.}, there exists a subset $S$ indexing the left singular vectors of:
  \[
    W^{\star}= M = U\operatorname{diag}(s_1,...s_{d_L},0,...,0)V^{\top}
  \]
  such that for $P_S:=U_SU_S^{\top}$ a linear projector onto the space of vectors indexed by the set $S$:
  \[
    \mu(\theta) := W = W_L...W_1 = P_S W^{\star} = U_SU_S^{\top}\Sigma_{YX}\Sigma_X^{-1}
  \]
\end{frame}

\begin{frame}{Takeaways}
  \begin{itemize}
    \item Loss landscapes are highly degenerate
    \item Under some conditions: local minima are global
    \item Discover geometric invariants by studying their symmetries
    \item A world of topology, geometry, group representation theory
  \end{itemize}
\end{frame}

\sectionframe{Learning Rate}

\begin{frame}{Descent lemma and stability}
  \small
  If $\mathcal{L}$ is convex and $\sigma$-smooth, then for gradient descent with any $\eta < \frac{2}{L}$,

  \[
    \mathcal{L}(\theta_{t+1})
    \le
    \mathcal{L}(\theta_t) -
    \eta\left(1-\frac{\eta \sigma}{2}\right)\|\nabla \mathcal{L}(\theta_t)\|_2^2
  \]

  Local stability near a minimum is controlled by the largest Hessian eigenvalue:
  \[
    \sigma \approx \lambda_{\max}(H)
    \qquad\Rightarrow\qquad
    \eta < \frac{2}{\lambda_{\max}(H)}
  \]

  Stability threshold ensuring monotonic decrease of the loss in convex optimization
\end{frame}

\begin{frame}{Edge of stability}
  The loss is non-convex, but the Hessian oscillates slightly above $\frac{2}{\eta}$: it regularizes away sharp minima

\sectionframe{Gradient flow}

\begin{frame}{Gradient flow and NTK dynamics}
  \small
  Small-learning-rate limit of gradient descent:
  \[
    \theta_{t+1}=\theta_t-\eta \nabla_\theta \mathcal{L}(\theta_t)
    \qquad \leadsto \qquad
    \dot{\theta}(t)=-\nabla_\theta \mathcal{L}(\theta(t))
  \]

  Neural Tangent Kernel (NTK):
  \[
    K(x,x',\theta) =
    \nabla_{\theta} f(\theta, x) \nabla_{\theta} f^{\top}(\theta, x) =
    \sum_{i=1}^p \partial_{\theta_i} f(\theta, x) \partial_{\theta_i} f^{\top}(\theta, x)
  \]

  For a quadratic loss on a dataset $(x_i,y_i)$, the dynamics in function space are
  \[
    \frac{d}{dt}f(\theta, x)
    = -\sum_{i=1}^N K_{\theta}(x,x_i)\big(f_{\theta}(x_i,t)-y_i\big)
  \]
\end{frame}

\begin{frame}{Conserved quantities}
  Conserved quantities under gradient flow:
  \[
    G_l = W_{l+1}^{\top}W_{l+1} - W_{l}W_{l}^{\top}
  \]

  When $G_l=0$ we have \textbf{balanced weights}

  \vspace{0.4cm}

  For a 1D, 2-layer DLN, we get a hyperbolic equation: $w_2^2 - w_1^2 = G$

  \vspace{0.4cm}

  See the tutorial for more
\end{frame}

\begin{frame}{Self-consistent equation in DLNs}
  \small
  Balanced weights $\implies$ self-consistent gradient-flow equation in function space:
  \[
    \dot{W} =
    \sum_{k=1}^{L}(WW^{\top})^{\frac{L-k}{L}}\nabla_W \mathcal{L}(W)(W^{\top}W)^{\frac{k-1}{L}}
  \]

  With operator $K: Q \mapsto \sum_{k=1}^{L}(WW^{\top})^{\frac{L-k}{L}} Q(W^{\top}W)^{\frac{k-1}{L}}$
  \[
    \dot{W} = K[\nabla\mathcal{L}(W)]
  \]
\end{frame}

\begin{frame}{Heuristic implicit regularization}
  \small
  \begin{itemize}
    \item Consider balanced weights of rank-$r$ students.
    \item Recall $\mu^{-1}(W)$ has many symmetries: rank patterns
    \item The singular values of weight $W^l$ are $\sigma^l_i$, and $p_k = \frac{\sigma^l_k}{\sum_k \sigma^l_k}$
    \item Spectral Entropy: $H^l(p)= - \sum_k p_k \log(p_k)$
    \item Entropy $S(W)=\sum_{l=1}^L H^l(p)$
    \item Free energy: $F(W) := L(W) - \frac{1}{\beta}S(W)$ with inverse temperature $\beta$
    \item Recall GF is $\dot{W}=-K[\nabla L(W)]$
    \item GF effectively minimizes a free energy functional, and entropy makes the regularization explicit
    \item For more, see \href{https://arxiv.org/abs/2411.09004}{The geometry of the deep linear network, Govind Menon}
  \end{itemize}
\end{frame}

\sectionframe{Initialization}

\begin{frame}{Lazy Regime (NTK regime)}
  \small
  For a DLN of depth $L$ with initialization $\theta(0)\sim N(0,\sigma)$ and $\sigma^2 = d_L^{-\gamma}, \ \gamma < 1$:
  \[
    \dot{W} \approx K_0[M - W]
  \]

  Leading to the following solution:
  \[
    W(t) = M - \exp(-K_0 t)[M - W(0)]
  \]

  The time scale of learning is fixed by the NTK at initialization:
  \[
    t \approx -k_0^{-1}\log\left(\frac{s - w_f}{s - w_0}\right)
  \]

  Behaves like a linear network with no simplicity bias
\end{frame}

\begin{frame}{Rich regime}
  \small
  For balanced and aligned weights (or $\gamma >1$), GF reduces to a system of 1D logistic ODEs:
  \[
    \dot{w}_{\alpha} = 2\left(s_{\alpha} - w_{\alpha}\right)w_{\alpha}.
  \]

  Solving this ODE for $w(t):=w_{\alpha}$ gives:
  \[
    w(t)=\frac{s}{1+\left(\frac{s}{w_0}-1\right)e^{-2s t}}.
  \]

  Timescale of learning
  \[
    t=\frac{1}{2s}\,\ln\!\left(\frac{w_f\,(s-w_0)}{w_0\,(s-w_f)}\right), \qquad s\neq 0.
  \]
\end{frame}

\sectionframe{Stochasticity}

\begin{frame}{Langevin dynamics}
  \begin{itemize}
    \item SGD: Sample a batch from a dataset and compute the batch gradient
    \item Continuous model with $B_t$ Brownian
  \end{itemize}

  \[
    d\theta_t=-\nabla L(\theta_t)\,dt+\sqrt{\eta\,\Sigma(\theta_t)}\,dB_t,
  \]

  \begin{itemize}
    \item Gradient noise covariance $\Sigma(\theta_t)$ depends on geometry: it is anisotropic and state-dependent
  \end{itemize}
\end{frame}

\begin{frame}{Fokker-Planck: Boltzmann equilibria}
  \small
  The Fokker-Planck equation (for density $p(\theta,t)$) is a local conservation law:
  \[
    \begin{aligned}
      \partial_t p & := - \nabla\cdot j \\
      j &:=- \nabla L(\theta)p(\theta) - \frac{\eta}{2}\nabla^\top\left(\Sigma(\theta)p(\theta)\right)
    \end{aligned}
  \]

  Important special case:
  \begin{itemize}
    \item Stationary distribution: $\partial_tp^{\ast}(x,t) = 0$
    \item Thermal equilibrium: $j=0$
    \item White noise: $\Sigma=\sigma^2 I$
  \end{itemize}

  \[
    p^{\ast}(\theta) \propto \exp\left(-\frac{2}{\eta\sigma^2}L_N(\theta)\right) \ \text{Boltzmann}
  \]
\end{frame}

\begin{frame}{Noise anisotropy}
  \small
  \begin{itemize}
    \item In general, noise is anisotropic: it introduces a different implicit bias
    \item In function space, Langevin introduces a drift
  \end{itemize}

  \[
    \frac{\eta}{2}\operatorname{Tr}(\Sigma(\theta)H(\theta))
  \]

  \begin{itemize}
    \item Heuristic: bias toward flatter minima
    \item More:
      \begin{itemize}
        \item see Govind Menon \href{https://arxiv.org/abs/2602.12257}{On the implicit regularization of Langevin dynamics with projected noise}
        \item My \href{https://arxiv.org/html/2604.06366v1}{paper} with collaborators deriving Langevin on DLNs during the saddle-to-saddle regime
      \end{itemize}
  \end{itemize}
\end{frame}
